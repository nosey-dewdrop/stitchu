import json, re
A = json.load(open('contract/anchors-v1.json'))
P = json.load(open('contract/primitives-v1.json'))
L = json.load(open('contract/edit-locality-v1.json'))
F = json.load(open('contract/figure-landmarks.json'))
E = json.load(open('vision/eval/live-2026-08-22.json'))

def words(s):  # camelCase + non-alpha -> lowercase word list
    return [w.lower() for w in re.split(r'[^A-Za-z]+', re.sub(r'(?<=[a-z])(?=[A-Z])', ' ', s)) if w]

# REFERANS KONUM LEKSIGI — V6-C md.4 KATMAN B'nin ayni kaynaklari, cipa
# sozlugunden BAGIMSIZ: her operatorun konum parametresinin degerleri +
# edit-locality bolgeleri + figure-landmarks. Cipa dosyasi bu kumeyi ETKILEMEZ.
lex = set()
for op in P['primitifler'].values():
    for k, v in (op.get('parametreler') or {}).items():
        if re.search('landmark|position|region', k, re.I) and isinstance(v.get('degerler'), list):
            for d in v['degerler']: lex |= set(words(d))
for z in L['zones']: lex |= set(words(z)) - {'zone'}
for k in F['landmarks']: lex |= set(words(k))
lex -= {'surface', 'global'}

# CIPA SOZLUGUNUN TASIDIGI KONUM SOZCUKLERI
anc = set()
for a in A['anchors']:
    anc |= set(words(a['ad'])) - {'zone'}

terms = []
for photo, r in E.items():
    for t in r.get('outOfVocab', []): terms.append((photo, t))
seen, uniq = set(), []
for p, t in terms:
    if t not in seen: seen.add(t); uniq.append((p, t))

konumlu = [(p, t, sorted(set(words(t)) & lex)) for p, t in uniq if set(words(t)) & lex]
tam = [x for x in konumlu if set(x[2]) <= anc]
eksik = [x for x in konumlu if not set(x[2]) <= anc]

print("serbest terim (outOfVocab, tekil): %d" % len(uniq))
print("referans konum leksigi: %d sozcuk = %s" % (len(lex), ' '.join(sorted(lex))))
print("cipa sozlugunun tasidigi konum sozcugu: %d = %s" % (len(anc), ' '.join(sorted(anc))))
print()
print("konum ibaresi TASIYAN terim: %d/%d" % (len(konumlu), len(uniq)))
print("cipa sozlugunun ibarenin TAMAMINI karsiladigi: %d/%d (konumlulara gore %d/%d)"
      % (len(tam), len(uniq), len(tam), len(konumlu)))
print()
print("-- KARSILANAN --")
for p, t, k in tam: print("  OK   %-42s konum=%s" % (t, ','.join(k)))
print("-- KARSILANMAYAN (konumlu ama cipa yok) --")
for p, t, k in eksik:
    print("  EKSIK %-41s konum=%s  ->  eksik=%s" % (t, ','.join(k), ','.join(sorted(set(k) - anc))))
print("-- KONUMSUZ (11 bekleniyor, V6-C md.5) --")
for p, t in uniq:
    if not set(words(t)) & lex: print("  --   %s" % t)
